<script setup lang="ts">
import { computed, ref } from "vue";

type Member = {
  id: string;
  name: string;
  role: "Designer" | "Frontend" | "Backend" | "PM";
  online: boolean;
  avatarUrl?: string;
};

const inputNeed = "__USER_NEED__";

const members = ref<Member[]>([
  { id: "m_1", name: "Ava", role: "Designer", online: true },
  { id: "m_2", name: "Noah", role: "Frontend", online: true },
  { id: "m_3", name: "Mia", role: "Backend", online: false },
  { id: "m_4", name: "Leo", role: "PM", online: false }
]);

const roleColor = (role: Member["role"]) => {
  switch (role) {
    case "Designer": return "bg-white/6 text-white/70 border-white/10";
    case "Frontend": return "bg-emerald-500/10 text-emerald-200 border-emerald-500/18";
    case "Backend": return "bg-sky-500/10 text-sky-200 border-sky-500/18";
    case "PM": return "bg-amber-500/10 text-amber-200 border-amber-500/18";
  }
};

const onlineCount = computed(() => members.value.filter(m => m.online).length);

function addMember() {
  members.value = [
    ...members.value,
    {
      id: `m_${members.value.length + 1}`,
      name: "New Member",
      role: "Frontend",
      online: true
    }
  ];
}
</script>

<template>
  <div class="min-h-screen bg-#0b0e14 text-white">
    <div class="mx-auto max-w-960 px-6 py-14">
      <div class="flex items-end justify-between gap-6">
        <div>
          <div class="text-xs text-white/55">
            Generated from input: {{ inputNeed }}
          </div>
          <h2 class="mt-3 text-2xl font-700 tracking-tight">
            Team Members
          </h2>
          <div class="mt-1 text-sm text-white/60">
            {{ onlineCount }} online · {{ members.length }} total
          </div>
        </div>

        <button
          class="rounded-xl border border-white/10 bg-white/8 px-4 py-2 text-sm font-600 hover:bg-white/12 transition"
          type="button"
          @click="addMember"
        >
          Add Member
        </button>
      </div>

      <div class="mt-7 grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
        <div
          v-for="m in members"
          :key="m.id"
          class="rounded-2xl border border-white/10 bg-white/4 px-4 py-4 hover:bg-white/6 hover:border-white/14 transition transform hover:-translate-y-0.5"
        >
          <div class="flex items-start justify-between gap-3">
            <div class="flex items-center gap-3 min-w-0">
              <div class="h-10 w-10 rounded-2xl border border-white/10 bg-white/7 grid place-items-center text-sm font-700">
                {{ m.name.slice(0, 1).toUpperCase() }}
              </div>
              <div class="min-w-0">
                <div class="text-sm font-650 truncate">
                  {{ m.name }}
                </div>
                <div class="mt-1 text-xs text-white/55">
                  Member ID: {{ m.id }}
                </div>
              </div>
            </div>
            <div class="flex items-center gap-2">
              <span class="h-2 w-2 rounded-full" :class="m.online ? 'bg-emerald-400' : 'bg-white/25'" />
            </div>
          </div>

          <div class="mt-4 flex items-center justify-between">
            <span class="inline-flex items-center rounded-full border px-2.5 py-1 text-xs" :class="roleColor(m.role)">
              {{ m.role }}
            </span>
            <span class="text-xs text-white/55">
              {{ m.online ? "Online" : "Offline" }}
            </span>
          </div>
        </div>
      </div>
    </div>
  </div>
</template>
